    30→/* ---------------- WhatsApp Bot Widget ---------------- */
    31→interface ChatMsg {
    32→  role: "user" | "bot";
    33→  content: string;
    34→}
    35→
    36→function WhatsappWidget() {
    37→  const [open, setOpen] = useState(false);
    38→  const [messages, setMessages] = useState<ChatMsg[]>([
    39→    {
    40→      role: "bot",
    41→      content:
    42→        "¡Hola! 👋 Soy el asistente virtual de ArteDur. ¿En qué te puedo ayudar? Puedo darte info de servicios, precios orientativos y coordinar tu presupuesto gratis 🏗️",
    43→    },
    44→  ]);
    45→  const [input, setInput] = useState("");
    46→  const [loading, setLoading] = useState(false);
    47→  const scrollRef = useRef<HTMLDivElement>(null);
    48→
    49→  useEffect(() => {
    50→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" });
    51→  }, [messages, loading]);
    52→
    53→  async function sendMessage() {
    54→    const text = input.trim();
    55→    if (!text || loading) return;
    56→    setInput("");
    57→    setMessages((m) => [...m, { role: "user", content: text }]);
    58→    setLoading(true);
    59→    try {
    60→      const res = await fetch("/api/whatsapp", {
    61→        method: "POST",
    62→        headers: { "Content-Type": "application/json" },
    63→        body: JSON.stringify({ phone: "web-widget", contactName: "Visitante Web", message: text }),
    64→      });
    65→      const data = await res.json();
    66→      setMessages((m) => [
    67→        ...m,
    68→        { role: "bot", content: data.reply || "Gracias por escribir! Te respondemos a la brevedad." },
    69→      ]);
    70→    } catch {
    71→      setMessages((m) => [
    72→        ...m,
    73→        { role: "bot", content: "Ups, hubo un error. Escribinos al 11-6691-9404 📱" },
    74→      ]);
    75→    } finally {
    76→      setLoading(false);
    77→    }
    78→  }
    79→
    80→  return (
    81→    <>
    82→      <button
    83→        onClick={() => setOpen(!open)}
    84→        aria-label="Chat WhatsApp"
    85→        className={`fixed bottom-5 right-5 z-50 flex h-14 w-14 items-center justify-center rounded-full shadow-2xl transition-all duration-300 ${
    86→          open ? "bg-[#1e1e1e] text-[#f5d000]" : "bg-[#25D366] text-white animate-pulse-yellow hover:scale-110"
    87→        }`}
    88→      >
    89→        {open ? <X className="h-6 w-6" /> : <MessageCircle className="h-7 w-7" />}
    90→      </button>
    91→
    92→      {open && (
    93→        <div className="fixed bottom-24 right-5 z-50 flex h-[480px] w-[calc(100vw-2.5rem)] max-w-[380px] flex-col overflow-hidden rounded-2xl border border-[#2b2b2b] bg-[#0f0f0f] shadow-2xl animate-fade-up">
    94→          <div className="flex items-center gap-3 bg-[#1e1e1e] px-4 py-3">
    95→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-[#25D366] text-white">
    96→              <MessageCircle className="h-5 w-5" />
    97→            </div>
    98→            <div>
    99→              <p className="text-sm font-semibold text-white">ArteDur · WhatsApp</p>
   100→              <p className="text-xs text-[#25D366]">● En línea · Respuesta inmediata</p>
   101→            </div>
   102→          </div>
   103→          <div ref={scrollRef} className="custom-scrollbar flex-1 space-y-3 overflow-y-auto bg-[#0b141a] p-4">
   104→            {messages.map((m, i) => (
   105→              <div key={i} className={`flex ${m.role === "user" ? "justify-end" : "justify-start"}`}>
   106→                <div
   107→                  className={`max-w-[85%] rounded-xl px-3 py-2 text-sm leading-relaxed ${
   108→                    m.role === "user"
   109→                      ? "rounded-br-sm bg-[#005c4b] text-white"
   110→                      : "rounded-bl-sm bg-[#1f2c33] text-[#e9edef]"
   111→                  }`}
   112→                >
   113→                  {m.content}
   114→                </div>
   115→              </div>
   116→            ))}
   117→            {loading && (
   118→              <div className="flex justify-start">
   119→                <div className="rounded-xl rounded-bl-sm bg-[#1f2c33] px-4 py-3">
   120→                  <div className="flex gap-1">
   121→                    <span className="h-2 w-2 animate-bounce rounded-full bg-[#f5d000]" style={{ animationDelay: "0ms" }} />
   122→                    <span className="h-2 w-2 animate-bounce rounded-full bg-[#f5d000]" style={{ animationDelay: "150ms" }} />
   123→                    <span className="h-2 w-2 animate-bounce rounded-full bg-[#f5d000]" style={{ animationDelay: "300ms" }} />
   124→                  </div>
   125→                </div>
   126→              </div>
   127→            )}
   128→          </div>
   129→          <div className="flex items-center gap-2 border-t border-[#2b2b2b] bg-[#1e1e1e] p-3">
   130→            <Input
   131→              value={input}
   132→              onChange={(e) => setInput(e.target.value)}
   133→              onKeyDown={(e) => e.key === "Enter" && sendMessage()}
   134→              placeholder="Escribí tu consulta..."
   135→              className="border-[#2b2b2b] bg-[#0f0f0f] text-white placeholder:text-[#666]"
   136→            />
   137→            <Button onClick={sendMessage} disabled={loading} size="icon" className="h-10 w-10 shrink-0 bg-[#25D366] text-white hover:bg-[#20bd5a]">
   138→              {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   139→            </Button>
   140→          </div>
   141→        </div>
   142→      )}
   143→    </>
   144→  );
   145→}

   580→/* ---------------- Contacto ---------------- */
   581→function ContactSection() {
   582→  const { toast } = useToast();
   583→  const [form, setForm] = useState({ name: "", phone: "", interest: "", message: "" });
   584→  const [sending, setSending] = useState(false);
   585→  const [sent, setSent] = useState(false);
   586→
   587→  async function handleSubmit(e: React.FormEvent) {
   588→    e.preventDefault();
   589→    if (!form.name.trim() || !form.message.trim()) {
   590→      toast({ title: "Faltan datos", description: "Completá al menos tu nombre y mensaje.", variant: "destructive" });
   591→      return;
   592→    }
   593→    setSending(true);
   594→    try {
   595→      const res = await fetch("/api/contact", {
   596→        method: "POST",
   597→        headers: { "Content-Type": "application/json" },
   598→        body: JSON.stringify(form),
   599→      });
   600→      if (!res.ok) throw new Error();
   601→      setSent(true);
   602→      setForm({ name: "", phone: "", interest: "", message: "" });
   603→      toast({
   604→        title: "¡Gracias! 🎉",
   605→        description: "Tu consulta fue recibida. Nuestro bot ya preparó una respuesta para vos.",
   606→      });
   607→    } catch {
   608→      toast({ title: "Error", description: "No se pudo enviar. Probá de nuevo.", variant: "destructive" });
   609→    } finally {
   610→      setSending(false);
   611→    }
   612→  }
   613→
   614→  return (
   615→    <section id="contacto" className="bg-[#141414] py-20 md:py-28">
   616→      <div className="mx-auto max-w-7xl px-4 sm:px-6">
   617→        <SectionHeader
   618→          tag="Estamos para ayudarte"
   619→          title="Contactanos"

   622→        <div className="grid gap-8 lg:grid-cols-2">

   624→            <div className="grid gap-4 sm:grid-cols-2">
   625→              <div className="space-y-2">
   626→                <Label htmlFor="c-name">Nombre completo *</Label>
   627→                <Input
   628→                  id="c-name"
   629→                  value={form.name}
   630→                  onChange={(e) => setForm({ ...form, name: e.target.value })}
